cmake_minimum_required(VERSION 3.20)
project(AetherPangoProvider LANGUAGES CXX)
option(AETHER_BUILD_PANGO_PROVIDER "Build experimental real Pango/Cairo provider" OFF)
if(AETHER_BUILD_PANGO_PROVIDER)
 find_package(PkgConfig REQUIRED)
 find_package(Threads REQUIRED)
 pkg_check_modules(PANGOCAIRO REQUIRED IMPORTED_TARGET pangocairo)
 if(NOT TARGET aether_native_preparation)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../preparation" "${CMAKE_CURRENT_BINARY_DIR}/preparation")
 endif()
 add_library(aether_pango_provider STATIC text_request.cpp pango_request_conversion.cpp pango_layout.cpp)
 target_compile_features(aether_pango_provider PUBLIC cxx_std_20)
 set_target_properties(aether_pango_provider PROPERTIES CXX_EXTENSIONS OFF)
 target_include_directories(aether_pango_provider PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(aether_pango_provider PUBLIC aether_native_preparation PkgConfig::PANGOCAIRO Threads::Threads)
endif()
