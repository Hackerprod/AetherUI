cmake_minimum_required(VERSION 3.20)
project(AetherCairoAdapter LANGUAGES CXX)
option(AETHER_BUILD_CAIRO_ADAPTER "Build experimental real Cairo renderer (requires Cairo SDK)" OFF)
if(AETHER_BUILD_CAIRO_ADAPTER)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
 if(NOT TARGET aether_native_preparation)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../preparation" "${CMAKE_CURRENT_BINARY_DIR}/preparation")
 endif()
 add_library(aether_cairo_adapter STATIC cairo_frame_renderer.cpp cairo_geometry.cpp ../native_frame.cpp)
 target_compile_features(aether_cairo_adapter PUBLIC cxx_std_20)
 set_target_properties(aether_cairo_adapter PROPERTIES CXX_EXTENSIONS OFF)
 target_include_directories(aether_cairo_adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(aether_cairo_adapter PUBLIC aether_native_preparation PRIVATE PkgConfig::CAIRO)
endif()
