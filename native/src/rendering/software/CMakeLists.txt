cmake_minimum_required(VERSION 3.20)
project(AetherSoftwareRaster LANGUAGES CXX)
if(NOT TARGET aether_native_preparation)
 add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../preparation" "${CMAKE_CURRENT_BINARY_DIR}/preparation")
endif()
add_library(aether_software_raster STATIC software_renderer.cpp ../native_frame.cpp)
target_compile_features(aether_software_raster PUBLIC cxx_std_20)
set_target_properties(aether_software_raster PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(aether_software_raster PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(aether_software_raster PUBLIC aether_native_preparation)
