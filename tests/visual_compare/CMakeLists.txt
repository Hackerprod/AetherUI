cmake_minimum_required(VERSION 3.20)
project(AetherVisualCompareContracts VERSION 0.1.0 LANGUAGES CXX)
add_library(aether_visual_compare_contract STATIC
 src/cli.cpp src/resource_manifest.cpp src/resource_validation.cpp
 src/report.cpp src/artifact_store.cpp src/sha256.cpp)
target_compile_features(aether_visual_compare_contract PUBLIC cxx_std_20)
set_target_properties(aether_visual_compare_contract PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(aether_visual_compare_contract PUBLIC include PRIVATE src
 "${CMAKE_CURRENT_SOURCE_DIR}/../../native/src/rendering/preparation/include")
add_executable(aether_visual_compare src/main.cpp)
target_link_libraries(aether_visual_compare PRIVATE aether_visual_compare_contract)
option(AETHER_VISUAL_COMPARE_BUILD_CONTRACT_TESTS "Build pure contract test executables (no automatic execution)" OFF)
if(AETHER_VISUAL_COMPARE_BUILD_CONTRACT_TESTS)
 foreach(name cli resource_manifest resource_validation report artifact_store sha256)
  add_executable(aether_visual_compare_test_${name} tests/test_${name}.cpp)
  target_link_libraries(aether_visual_compare_test_${name} PRIVATE aether_visual_compare_contract)
  target_include_directories(aether_visual_compare_test_${name} PRIVATE src)
 endforeach()
endif()

option(AETHER_VISUAL_COMPARE_BUILD_PREPARATION_TESTS "Build experimental preparation library and synthetic tests (no automatic execution)" OFF)
if(AETHER_VISUAL_COMPARE_BUILD_PREPARATION_TESTS)
 set(aether_preparation_abi_include "${CMAKE_CURRENT_SOURCE_DIR}/../../native/include")
 if(NOT EXISTS "${aether_preparation_abi_include}/aether/aether.h")
  message(FATAL_ERROR "Preparation requires the public aether/aether.h from this checkout")
 endif()
 add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../native/src/rendering/preparation"
  "${CMAKE_CURRENT_BINARY_DIR}/native_preparation")
 add_library(aether_visual_compare_preparation INTERFACE)
 target_include_directories(aether_visual_compare_preparation INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
 target_link_libraries(aether_visual_compare_preparation INTERFACE aether_native_preparation)
 foreach(name aether_command_source paint_snapshot preparation_metrics preparation_compat)
  add_executable(aether_visual_compare_test_${name} tests/test_${name}.cpp)
  target_include_directories(aether_visual_compare_test_${name} PRIVATE "${aether_preparation_abi_include}")
  target_link_libraries(aether_visual_compare_test_${name} PRIVATE aether_visual_compare_preparation)
  set_target_properties(aether_visual_compare_test_${name} PROPERTIES CXX_EXTENSIONS OFF)
 endforeach()
endif()
