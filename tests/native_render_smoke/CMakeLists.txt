cmake_minimum_required(VERSION 3.20)
project(AetherNativeRenderSmoke LANGUAGES CXX)
option(AETHER_BUILD_NATIVE_RENDER_SMOKE "Build real SDK smoke runner (no automatic execution)" OFF)
if(AETHER_BUILD_NATIVE_RENDER_SMOKE)
 if(NOT AETHER_BUILD_CAIRO_ADAPTER OR NOT AETHER_BUILD_PANGO_PROVIDER)
  message(FATAL_ERROR "Explicitly enable AETHER_BUILD_CAIRO_ADAPTER and AETHER_BUILD_PANGO_PROVIDER")
 endif()
 add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../native/src/rendering/cairo" "${CMAKE_CURRENT_BINARY_DIR}/cairo")
 add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../native/src/rendering/text" "${CMAKE_CURRENT_BINARY_DIR}/text")
 add_executable(aether_native_render_smoke main.cpp fixture.cpp report.cpp)
 target_compile_features(aether_native_render_smoke PRIVATE cxx_std_20)
 set_target_properties(aether_native_render_smoke PROPERTIES CXX_EXTENSIONS OFF)
 target_link_libraries(aether_native_render_smoke PRIVATE aether_cairo_adapter aether_pango_provider)
endif()
